// OAuth flow state — single-use, expiring, anti-CSRF + anti-replay.
// - `stateHash`: SHA-256 of the raw `state` token (state appears in URLs and
//   logs, so we hash it at rest).
// - `codeVerifier`: the raw PKCE code_verifier (256-bit random, base64url).
//   Stored raw because (a) it is single-use, (b) it expires in 10 minutes,
//   (c) it is only useful alongside our client_secret (env, not in DB), and
//   (d) we need the raw value at token-exchange time. Redeeming an auth code
//   requires the auth code + redirect_uri + code_verifier + client_secret —
//   leaking the verifier alone is insufficient. See SECURITY_MODEL.md §"OAuth".
model OAuthState {
  id                String    @id @default(cuid())
  stateHash         String    @unique
  codeVerifier      String
  provider          String    // "google" | "github"
  redirectTarget    String?   // post-login path (e.g. "/app")
  ipHash            String?
  userAgentSummary  String?
  createdAt         DateTime  @default(now())
  expiresAt         DateTime
  usedAt            DateTime?

  @@index([provider])
  @@index([expiresAt])
}

// Project-scoped secret vault for {{secret.NAME}} references in journey steps.
// Values are AES-256-GCM encrypted with the master key. Keys are returned to the UI;
// values are only ever decrypted inside the worker (never serialized to API responses).

  id              String   @id @default(cuid())
  projectId       String
  key             String   // [a-zA-Z0-9_]+, max 64 chars
  valueEncrypted  String   // JSON EncryptedValue
  description     String?
  createdById     String?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  project         Project  @relation(fields: [projectId], references: [id], onDelete: Cascade)
  createdBy       User?    @relation("ProjectSecretCreator", fields: [createdById], references: [id], onDelete: SetNull)

  @@unique([projectId, key])
  @@index([projectId])
}

  id              String   @id @default(cuid())
  projectId       String

  // Phase 8 — AI semantic finding grouping. Holds the structured
  // SemanticGrouping JSON ({ groups: [{ groupId, label, findingIds[],
  // sharedRootCause }] }) so reports can present "one underlying issue"
  // instead of listing every duplicate individually. Null until generated.
  aiSemanticGroupingJson String?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  environment     ProjectEnvironment? @relation(fields: [environmentId], references: [id], onDelete: SetNull)
  scanProfile     ScanProfile? @relation(fields: [scanProfileId], references: [id], onDelete: SetNull)
  triggeredBy     User?    @relation(fields: [triggeredById], references: [id], onDelete: SetNull)
