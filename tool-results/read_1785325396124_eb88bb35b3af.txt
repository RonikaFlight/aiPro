   147→function relativeTime(date: string): string {
   148→  const diff = Date.now() - new Date(date).getTime()
   149→  const mins = Math.floor(diff / 60_000)
   150→  if (mins < 1) return 'just now'
   151→  if (mins < 60) return `${mins}m ago`
   152→  const hours = Math.floor(mins / 60)
   153→  if (hours < 24) return `${hours}h ago`
   154→  const days = Math.floor(hours / 24)
   155→  if (days < 30) return `${days}d ago`
   156→  return new Date(date).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   157→}

   159→function formatScore(score: number | null): string {
   160→  if (score === null || score === undefined) return '—'
   161→  return `${Math.round(score * 100)}%`
   162→}

   164→function statusColor(status: string): string {
   165→  switch (status) {
   166→    case 'COMPLETED':
   167→      return 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300'
   168→    case 'COMPLETED_WITH_WARNINGS':
   169→      return 'bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300'
   170→    case 'FAILED':
   171→      return 'bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300'
   172→    case 'CANCELLED':
   173→    case 'TIMED_OUT':
   174→      return 'bg-gray-100 text-gray-800 dark:bg-gray-800 dark:text-gray-300'
   175→    case 'RUNNING':
   176→    case 'ANALYZING':
   177→      return 'bg-blue-100 text-blue-800 dark:bg-blue-900/40 dark:text-blue-300'
   178→    default:
   179→      return 'bg-gray-100 text-gray-800 dark:bg-gray-800 dark:text-gray-300'
   180→  }
   181→}
   182→
   183→function readinessColor(readiness: string): string {
   184→  switch (readiness) {
   185→    case 'READY':
   186→      return 'text-emerald-600 dark:text-emerald-400'
   187→    case 'NEEDS_WORK':
   188→      return 'text-amber-600 dark:text-amber-400'
   189→    case 'NOT_READY':
   190→      return 'text-red-600 dark:text-red-400'
   191→    default:
   192→      return 'text-muted-foreground'
   193→  }
   194→}
   195→
   196→function readinessBg(readiness: string): string {
   197→  switch (readiness) {
   198→    case 'READY':
   199→      return 'bg-emerald-50 border-emerald-200 dark:bg-emerald-900/20 dark:border-emerald-800'
   200→    case 'NEEDS_WORK':
   201→      return 'bg-amber-50 border-amber-200 dark:bg-amber-900/20 dark:border-amber-800'
   202→    case 'NOT_READY':
   203→      return 'bg-red-50 border-red-200 dark:bg-red-900/20 dark:border-red-800'
   204→    default:
   205→      return 'bg-muted border-border'
   206→  }
   207→}

   209→function gradeColor(grade: string): string {
   210→  switch (grade) {
   211→    case 'A': return 'bg-emerald-600'
   212→    case 'B': return 'bg-lime-600'
   213→    case 'C': return 'bg-amber-500'
   214→    case 'D': return 'bg-orange-500'
   215→    case 'F': return 'bg-red-600'
   216→    default: return 'bg-gray-500'
   217→  }
   218→}

   220→function severityColor(severity: string): string {
   221→  switch (severity) {
   222→    case 'BLOCKER':
   223→      return 'bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300'
   224→    case 'CRITICAL':
   225→      return 'bg-orange-100 text-orange-800 dark:bg-orange-900/40 dark:text-orange-300'
   226→    case 'MAJOR':
   227→      return 'bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300'
   228→    case 'MINOR':
   229→      return 'bg-lime-100 text-lime-800 dark:bg-lime-900/40 dark:text-lime-300'
   230→    case 'INFO':
   231→      return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   232→    default:
   233→      return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   234→  }
   235→}

   237→function scoreTrendIcon(current: number | null, previous: number | null) {
   238→  if (current === null || previous === null) return <Minus className="h-4 w-4 text-muted-foreground" />
   239→  if (current > previous) return <TrendingUp className="h-4 w-4 text-emerald-600" />
   240→  if (current < previous) return <TrendingDown className="h-4 w-4 text-red-500" />
   241→  return <Minus className="h-4 w-4 text-muted-foreground" />
   242→}

   606→          {/* By Category */}
   607→          <Card>

   616→                </div>

   618→                <div className="space-y-2 max-h-64 overflow-y-auto">
   619→                  {findingsByCategory.map((g) => {
   620→                    const maxCount = findingsByCategory[0]?.count ?? 1
   621→                    const pct = Math.round((g.count / maxCount) * 100)
   622→                    return (
   623→                      <div key={g.category} className="space-y-1">
   624→                        <div className="flex items-center justify-between text-sm">

   627→                        </div>
   628→                        <Progress value={pct} className="h-1.5" />

   633→              )}
   634→            </CardContent>
   635→          </Card>

   642→            <CardHeader>

   676→
   677→          {/* Coverage */}
   678→          <Card className="lg:col-span-1">
   679→            <CardHeader>

   681→                <Globe className="h-4 w-4" />
   682→                Coverage
   683→              </CardTitle>
   684→              <CardDescription>Tested environments and configurations</CardDescription>
   685→            </CardHeader>
   686→            <CardContent>
   687→              <div className="space-y-4">
   688→                <div>
   689→                  <div className="flex items-center gap-2 text-sm mb-2">

   703→                    </div>

   705→                    <span className="text-sm text-muted-foreground">Not yet scanned</span>

   708→                <div>
   709→                  <div className="text-xs text-muted-foreground mb-1.5">Viewports</div>
   710→                  {coverage.viewports.length > 0 ? (
   711→                    <div className="flex flex-wrap gap-1.5">
   712→                      {coverage.viewports.map((v) => (
   713→                        <Badge key={v} variant="secondary" className="text-xs">{v}</Badge>

   715→                    </div>

   717→                    <span className="text-sm text-muted-foreground">Not yet scanned</span>

   740→                <Route className="h-4 w-4" />

   781→                    </div>

   783→                      <div className="font-bold tabular-nums">{journeys.activeJourneys}</div>
   784→                      <div className="text-xs text-muted-foreground">Active</div>
   785→                    </div>
   786→                  </div>
   787→                </div>
   788→              )}
   789→            </CardContent>

   794→        <Card className="mb-6">

   804→                </Link>

   806→            </div>

   808→          <CardContent>
   809→            {recentReports.length === 0 ? (

   841→              </div>

   845→      </main>
