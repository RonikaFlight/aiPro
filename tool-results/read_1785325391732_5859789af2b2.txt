   126→function relativeTime(date: string): string {
   127→  const diff = Date.now() - new Date(date).getTime()
   128→  const mins = Math.floor(diff / 60_000)
   129→  if (mins < 1) return 'just now'
   130→  if (mins < 60) return `${mins}m ago`
   131→  const hours = Math.floor(mins / 60)
   132→  if (hours < 24) return `${hours}h ago`
   133→  const days = Math.floor(hours / 24)
   134→  if (days < 30) return `${days}d ago`
   135→  return new Date(date).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   136→}

   138→function severityColor(severity: string): string {
   139→  switch (severity) {
   140→    case 'BLOCKER':
   141→      return 'bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300 border-red-200 dark:border-red-800'
   142→    case 'CRITICAL':
   143→      return 'bg-orange-100 text-orange-800 dark:bg-orange-900/40 dark:text-orange-300 border-orange-200 dark:border-orange-800'
   144→    case 'MAJOR':
   145→      return 'bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300 border-amber-200 dark:border-amber-800'
   146→    case 'MINOR':
   147→      return 'bg-lime-100 text-lime-800 dark:bg-lime-900/40 dark:text-lime-300 border-lime-200 dark:border-lime-800'
   148→    case 'INFO':
   149→      return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300 border-gray-200 dark:border-gray-700'
   150→    default:
   151→      return 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   152→  }
   153→}

   178→function formatStatus(status: string): string {
   179→  return status
   180→    .replace(/_/g, ' ')
   181→    .replace(/\b\w/g, (c) => c.toUpperCase())
   182→}

   184→function truncateUrl(url: string, maxLen = 60): string {
   185→  if (url.length <= maxLen) return url
   186→  return url.slice(0, maxLen - 3) + '...'
   187→}

   191→export default function FindingsTablePage() {
   192→  const params = useParams<{ projectId: string }>()
   193→  const projectId = params.projectId
   194→
   195→  const [findings, setFindings] = useState<FindingsResponse | null>(null)
   196→  const [project, setProject] = useState<ProjectBrief | null>(null)
   197→  const [loading, setLoading] = useState(true)
   198→  const [error, setError] = useState<string | null>(null)
   199→
   200→  // Filters
   201→  const [selectedSeverities, setSelectedSeverities] = useState<string[]>([])
   202→  const [selectedStatuses, setSelectedStatuses] = useState<string[]>([])

   204→  const [searchText, setSearchText] = useState('')
   205→  const [searchInput, setSearchInput] = useState('')
   206→
   207→  // Sort
   208→  const [sortField, setSortField] = useState<SortField>('lastSeenAt')
   209→  const [sortOrder, setSortOrder] = useState<SortOrder>('desc')
   210→
   211→  // Pagination
   212→  const [cursor, setCursor] = useState<string | undefined>(undefined)
   213→  const [prevCursors, setPrevCursors] = useState<string[]>([])
   214→
   215→  // Available categories derived from loaded findings
   216→  const [availableCategories, setAvailableCategories] = useState<string[]>([])
   217→
   218→  // ─── Data Fetching ────────────────────────────────────────
   219→
   220→  const buildQueryString = useCallback(() => {
   221→    const sp = new URLSearchParams()
   222→    sp.set('limit', PAGE_SIZE.toString())
   223→    if (sortField) sp.set('sort', sortField)
   224→    if (sortOrder) sp.set('order', sortOrder)
   225→    if (cursor) sp.set('cursor', cursor)
   226→    if (selectedSeverities.length > 0) sp.set('severity', selectedSeverities.join(','))
   227→    if (selectedStatuses.length > 0) sp.set('status', selectedStatuses.join(','))
   228→    if (category) sp.set('category', category)
   229→    if (searchText) sp.set('search', searchText)
   230→    return sp.toString()
   231→  }, [sortField, sortOrder, cursor, selectedSeverities, selectedStatuses, category, searchText])
   232→
   233→  const loadFindings = useCallback(async () => {
   234→    setLoading(true)
   235→    setError(null)
   236→    try {
   237→      const qs = buildQueryString()
   238→      const res = await fetch(`/api/v1/projects/${projectId}/findings?${qs}`)
   239→      if (!res.ok) {
   240→        const body = await res.json().catch(() => ({}))
   241→        setError(body.detail || `Failed to load findings (${res.status})`)
   242→        return
   243→      }
   244→      const data = (await res.json()) as FindingsResponse
   245→      setFindings(data)
   246→
   247→      // Derive unique categories
   248→      const cats = new Set<string>()
   249→      data.items.forEach((f) => cats.add(f.category))
   250→      setAvailableCategories((prev) => {
   251→        const merged = new Set([...prev, ...cats])
   252→        return Array.from(merged).sort()
   253→      })
   254→    } catch (err) {
   255→      setError(err instanceof Error ? err.message : 'Failed to load')
   256→    } finally {
   257→      setLoading(false)
   258→    }
   259→  }, [projectId, buildQueryString])
   260→
   261→  const loadProject = useCallback(async () => {
   262→    try {
   263→      const res = await fetch(`/api/v1/projects/${projectId}`)
   264→      if (res.ok) {
   265→        setProject(await res.json())
   266→      }
   267→    } catch {
   268→      // Non-critical; breadcrumb can still work without it
   269→    }
   270→  }, [projectId])
   271→
   272→  useEffect(() => {
   273→    void loadProject()
   274→  }, [loadProject])
   275→
   276→  useEffect(() => {
   277→    void loadFindings()
   278→  }, [loadFindings])
   279→
   280→  // Reset pagination when filters/sort change
   281→  useEffect(() => {
   282→    setCursor(undefined)
   283→    setPrevCursors([])
   284→  }, [selectedSeverities, selectedStatuses, category, searchText, sortField, sortOrder])
   285→
   286→  // ─── Handlers ──────────────────────────────────────────────
   287→
   288→  const handleSearch = () => {
   289→    setSearchText(searchInput)
   290→  }
   291→
   292→  const handleSearchKeyDown = (e: React.KeyboardEvent) => {
   293→    if (e.key === 'Enter') {
   294→      handleSearch()
   295→    }
   296→  }
   297→
   298→  const toggleSeverity = (sev: string) => {
   299→    setSelectedSeverities((prev) =>
   300→      prev.includes(sev) ? prev.filter((s) => s !== sev) : [...prev, sev],
   301→    )
   302→  }
   303→
   304→  const toggleStatus = (status: string) => {
   305→    setSelectedStatuses((prev) =>
   306→      prev.includes(status) ? prev.filter((s) => s !== status) : [...prev, status],
   307→    )
   308→  }
   309→
   310→  const toggleSortOrder = () => {
   311→    setSortOrder((prev) => (prev === 'desc' ? 'asc' : 'desc'))
   312→  }
   313→
   314→  const goNext = () => {
   315→    if (findings?.nextCursor) {
   316→      setPrevCursors((prev) => [...prev, cursor ?? ''])
   317→      setCursor(findings.nextCursor)
   318→    }
   319→  }
   320→
   321→  const goPrev = () => {
   322→    if (prevCursors.length > 0) {
   323→      const newPrev = [...prevCursors]
   324→      const prevCursor = newPrev.pop()!
   325→      setPrevCursors(newPrev)
   326→      setCursor(prevCursor || undefined)
   327→    }
   328→  }
   329→
   330→  const clearFilters = () => {
   331→    setSelectedSeverities([])
   332→    setSelectedStatuses([])
   333→    setCategory('')
   334→    setSearchText('')
   335→    setSearchInput('')
   336→  }
   337→
   338→  const hasActiveFilters =
   339→    selectedSeverities.length > 0 ||
   340→    selectedStatuses.length > 0 ||
   341→    category !== '' ||
   342→    searchText !== ''
   343→
   344→  const activeFilterCount =
   345→    selectedSeverities.length + selectedStatuses.length + (category ? 1 : 0) + (searchText ? 1 : 0)
   346→
   347→  // ─── Loading Skeleton ───────────────────────────────────────
   348→
   349→  if (loading && !findings) {
   350→    return (
   351→      <div className="min-h-screen flex flex-col bg-background">
   352→        <PageHeader projectName={project?.name} />
   353→        <main className="flex-1 mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8 py-8">
   354→          <div className="animate-pulse space-y-6">
   355→            <Skeleton className="h-4 w-64" />
   356→            <Skeleton className="h-10 w-full rounded-lg" />
   357→            <Skeleton className="h-96 w-full rounded-lg" />
   358→          </div>
   359→        </main>
   360→      </div>
   361→    )
   362→  }
   363→
   364→  // ─── Error State ───────────────────────────────────────────
   365→
   366→  if (error && !findings) {
   367→    return (
   368→      <div className="min-h-screen flex flex-col bg-background">
   369→        <PageHeader projectName={project?.name} />
   370→        <main className="flex-1 mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8 py-8">
   371→          <Card>
   372→            <CardHeader>
   373→              <CardTitle>Failed to load findings</CardTitle>
   374→              <CardDescription>{error}</CardDescription>
   375→            </CardHeader>
   376→            <CardContent>
   377→              <Button onClick={() => void loadFindings()} variant="outline">
   378→                Retry
   379→              </Button>
   380→            </CardContent>
   381→          </Card>
   382→        </main>
   383→      </div>
   384→    )
   385→  }
   386→
   387→  // ─── Main Render ────────────────────────────────────────────
   388→
   389→  const canGoPrev = prevCursors.length > 0
   390→  const canGoNext = !!findings?.nextCursor
   391→  const totalItems = findings?.totalApprox ?? 0
   392→
   393→  return (
   394→    <div className="min-h-screen flex flex-col bg-background">
   395→      <PageHeader projectName={project?.name} />
   396→
   397→      <main className="flex-1 mx-auto w-full max-w-7xl px-4 sm:px-6 lg:px-8 py-8">
   398→        {/* Breadcrumb */}
   399→        <div className="flex items-center gap-2 text-sm text-muted-foreground mb-6">
   400→          <Link href="/app" className="hover:text-foreground transition-colors">
   401→            Dashboard
   402→          </Link>
   403→          <ChevronRight className="h-3.5 w-3.5" />
   404→          {project && (
   405→            <>
   406→              <Link href={`/app/projects/${projectId}`} className="hover:text-foreground transition-colors">
   407→                {project.name}
   408→              </Link>
   409→              <ChevronRight className="h-3.5 w-3.5" />
   410→            </>
   411→          )}
   412→          <span className="text-foreground font-medium">Findings</span>
   413→        </div>
   414→
   415→        {/* Page Title */}
   416→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 mb-6">
   417→          <div className="flex items-center gap-3">
   418→            <Button variant="ghost" size="icon" asChild>
   419→              <Link href={`/app/projects/${projectId}`}>
   420→                <ArrowLeft className="h-4 w-4" />
   421→              </Link>
   422→            </Button>
   423→            <h1 className="text-2xl sm:text-3xl font-bold tracking-tight">Findings</h1>

   430→        {/* Filters Bar */}
   431→        <Card className="mb-6">
