   102→function getCsrfToken(): Promise<string> {

   106→function statusColor(status: string) {
   107→  switch (status) {
   108→    case 'ACTIVE':
   109→      return 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300 border-emerald-200 dark:border-emerald-800'
   110→    case 'SUSPENDED':
   111→      return 'bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-300 border-red-200 dark:border-red-800'
   112→    case 'DELETED':
   113→      return 'bg-gray-100 text-gray-600 dark:bg-gray-800/40 dark:text-gray-400 border-gray-200 dark:border-gray-700'
   114→    case 'PENDING_VERIFICATION':
   115→      return 'bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-300 border-amber-200 dark:border-amber-800'
   116→    default:
   117→      return 'bg-gray-100 text-gray-600 dark:bg-gray-800/40 dark:text-gray-400 border-gray-200 dark:border-gray-700'
   118→  }
   119→}

   121→function roleColor(role: string) {
   122→  switch (role) {

   124→      return 'bg-violet-100 text-violet-800 dark:bg-violet-900/40 dark:text-violet-300 border-violet-200 dark:border-violet-800'

   133→function formatRole(role: string) {
   134→  switch (role) {

   136→    case 'SUPPORT': return 'Support'
   137→    case 'USER': return 'User'
   138→    default: return role
   139→  }

   142→function formatStatus(status: string) {

   146→function relativeTime(dateStr: string | null): string {
   147→  if (!dateStr) return '\u2014'
   148→  const date = new Date(dateStr)
   149→  const now = new Date()
   150→  const diffMs = now.getTime() - date.getTime()
   151→  const diffSec = Math.floor(diffMs / 1000)
   152→  const diffMin = Math.floor(diffSec / 60)
   153→  const diffHr = Math.floor(diffMin / 60)
   154→  const diffDay = Math.floor(diffHr / 24)
   155→
   156→  if (diffSec < 60) return 'just now'
   157→  if (diffMin < 60) return `${diffMin}m ago`
   158→  if (diffHr < 24) return `${diffHr}h ago`
   159→  if (diffDay < 30) return `${diffDay}d ago`
   160→  return date.toLocaleDateString()
   161→}
   162→
   163→function formatDate(dateStr: string): string {
   164→  return new Date(dateStr).toLocaleDateString(undefined, {
   165→    year: 'numeric',
   166→    month: 'short',
   167→    day: 'numeric',
   168→  })
   169→}
