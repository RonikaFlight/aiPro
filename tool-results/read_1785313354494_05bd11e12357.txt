   146→   146→function relativeTime(dateStr: string | null): string {
   147→   147→  if (!dateStr) return '\u2014'
   148→   148→  const date = new Date(dateStr)
   149→   149→  const now = new Date()
   150→   150→  const diffMs = now.getTime() - date.getTime()
   151→   151→  const diffSec = Math.floor(diffMs / 1000)
   152→   152→  const diffMin = Math.floor(diffSec / 60)
   153→   153→  const diffHr = Math.floor(diffMin / 60)
   154→   154→  const diffDay = Math.floor(diffHr / 24)
   155→   155→
   156→   156→  if (diffSec < 60) return 'just now'
   157→   157→  if (diffMin < 60) return `${diffMin}m ago`
   158→   158→  if (diffHr < 24) return `${diffHr}h ago`
   159→   159→  if (diffDay < 30) return `${diffDay}d ago`
   160→   160→  return date.toLocaleDateString()
   161→   161→}

   166→   166→    month: 'short',
   167→   167→    day: 'numeric',
   168→   168→  })
   169→   169→}
