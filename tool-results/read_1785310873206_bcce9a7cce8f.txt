   146→   146→// OAuth flow state — single-use, expiring, anti-CSRF + anti-replay.
   147→   147→// - `stateHash`: SHA-256 of the raw `state` token (state appears in URLs and
   148→   148→//   logs, so we hash it at rest).
   149→   149→// - `codeVerifier`: the raw PKCE code_verifier (256-bit random, base64url).
   150→   150→//   Stored raw because (a) it is single-use, (b) it expires in 10 minutes,
   151→   151→//   (c) it is only useful alongside our client_secret (env, not in DB), and
   152→   152→//   (d) we need the raw value at token-exchange time. Redeeming an auth code
   153→   153→//   requires the auth code + redirect_uri + code_verifier + client_secret —
   154→   154→//   leaking the verifier alone is insufficient. See SECURITY_MODEL.md §"OAuth".
   155→   155→model OAuthState {
   156→   156→  id                String    @id @default(cuid())
   157→   157→  stateHash         String    @unique
   158→   158→  codeVerifier      String
   159→   159→  provider          String    // "google" | "github"
   160→   160→  redirectTarget    String?   // post-login path (e.g. "/app")
   161→   161→  ipHash            String?
   162→   162→  userAgentSummary  String?
   163→   163→  createdAt         DateTime  @default(now())
   164→   164→  expiresAt         DateTime
   165→   165→  usedAt            DateTime?
   166→   166→
   167→   167→  @@index([provider])
   168→   168→  @@index([expiresAt])
   169→   169→}
