   145→   145→// OAuth flow state — single-use, expiring, anti-CSRF + anti-replay.
   146→   146→// - `stateHash`: SHA-256 of the raw `state` token (state appears in URLs and
   147→   147→//   logs, so we hash it at rest).
   148→   148→// - `codeVerifier`: the raw PKCE code_verifier (256-bit random, base64url).
   149→   149→//   Stored raw because (a) it is single-use, (b) it expires in 10 minutes,
   150→   150→//   (c) it is only useful alongside our client_secret (env, not in DB), and
   151→   151→//   (d) we need the raw value at token-exchange time. Redeeming an auth code
   152→   152→//   requires the auth code + redirect_uri + code_verifier + client_secret —
   153→   153→//   leaking the verifier alone is insufficient. See SECURITY_MODEL.md §"OAuth".
   154→   154→model OAuthState {
   155→   155→  id                String    @id @default(cuid())
   156→   156→  stateHash         String    @unique
   157→   157→  codeVerifier      String
   158→   158→  provider          String    // "google" | "github"
   159→   159→  redirectTarget    String?   // post-login path (e.g. "/app")
   160→   160→  ipHash            String?
   161→   161→  userAgentSummary  String?
   162→   162→  createdAt         DateTime  @default(now())
   163→   163→  expiresAt         DateTime
   164→   164→  usedAt            DateTime?
   165→   165→
   166→   166→  @@index([provider])
   167→   167→  @@index([expiresAt])
   168→   168→}
